/*************************************************
 * HFD Pre-Plan Web App — Save + Image Proxy + Upload (JSONP friendly)
 * Endpoints
 *   POST  /exec:  fn=save&payload=<json>
 *   GET   /exec:  ?fn=save&callback=cb&payload=<json>     (JSONP)
 *   GET   /exec:  ?id=<driveFileId>[&w=800]  or  ?fn=img&u=<https-url>[&w=800]
 *   GET   /exec:  ?fn=imgpage&id=<driveFileId>[&w=800]    (HTML test page)
 *   GET   /exec:  ?fn=imgdebug&id=<driveFileId>[&w=800]   (debug JSON: which path used)
 *   POST  /exec:  fn=photo&payload={"photo":"data:image/jpeg;base64,...","name":"opt","folderId":"opt"}
 *   GET   /exec:  ?fn=ping[&callback=cb]
 *   OPTIONS /exec: preflight (no headers set)
 *************************************************/

/** ===== CONFIG ===== */
var SPREADSHEET_ID = '1B1JbGtXAl-ymNQ-vb4UBRwZZ36byfu3fva0XpstidNA';
var SHEET_TAB      = 'Form Responses';
var KEY_COL_HEADER = 'Stable ID';
var ALLOW_APPEND   = true;
var PHOTO_FOLDER_ID = '';

/** ===== JSON / JSONP responder ===== */
function respond_(e, obj) {
  var cb = (e && e.parameter && e.parameter.callback) ? String(e.parameter.callback) : null;
  if (cb) {
    var jp = ContentService.createTextOutput(cb + '(' + JSON.stringify(obj || {}) + ')');
    jp.setMimeType(ContentService.MimeType.JAVASCRIPT);
    return jp;
  }
  var js = ContentService.createTextOutput(JSON.stringify(obj || {}));
  js.setMimeType(ContentService.MimeType.JSON);
  return js;
}

/** ===== Diagnostics ===== */
function who_(e) {
  return respond_(e, {
    ok: true,
    effectiveUser: (Session.getEffectiveUser() && Session.getEffectiveUser().getEmail()) || '(blank)',
    activeUser: (Session.getActiveUser() && Session.getActiveUser().getEmail()) || '(blank)'
  });
}
function check_(e) {
  try {
    SpreadsheetApp.openById(SPREADSHEET_ID);
    return respond_(e, { ok: true, canOpen: true, spreadsheetId: SPREADSHEET_ID });
  } catch (err) {
    return respond_(e, { ok: false, canOpen: false, error: String(err), spreadsheetId: SPREADSHEET_ID });
  }
}

/** ===== Entrypoints ===== */
function doOptions(e) { return respond_(e, { ok: true }); }
function doPost(e) {
  try {
    var fn = ((e && e.parameter && e.parameter.fn) || '').toLowerCase();
    if (fn === 'save')  return handleSave_(e);
    if (fn === 'photo') return handlePhotoUpload_(e);
    return respond_(e, { ok:false, error:'unsupported-fn' });
  } catch (err) { return respond_(e, { ok:false, error:String(err) }); }
}
function doGet(e) {
  try {
    var fn = ((e && e.parameter && e.parameter.fn) || '').toLowerCase();
    if (fn === 'who')     return who_(e);
    if (fn === 'check')   return check_(e);
    if (fn === 'ping')    return respond_(e, { ok:true, pong:'ok' });
    if (fn === 'save')    return handleSave_(e);
    if (fn === 'imgpage') return imgPage_(e);
    if (fn === 'imgdebug')return imgDebug_(e);
    if (fn === 'img' || (e.parameter && (e.parameter.id || e.parameter.u))) {
      return handleImageProxy_(e);
    }
    return respond_(e, { ok:true, ping:'pong' });
  } catch (err) { return respond_(e, { ok:false, error:String(err) }); }
}

/** ===== SAVE (by Stable ID) ===== */
function handleSave_(e) {
  var payload = parsePayload_(e);
  if (!payload || typeof payload !== 'object') return respond_(e, { ok:false, error:'missing-payload' });

  var key = getKey_(payload);
  if (!key) return respond_(e, { ok:false, error:'missing-key: ' + KEY_COL_HEADER });

  var sh = getSheet_();
  if (!sh) return respond_(e, { ok:false, error:'sheet-not-found' });

  // Ensure header row exists
  if (sh.getLastRow() < 1) {
    var hdrs = Object.keys(payload);
    if (hdrs.indexOf(KEY_COL_HEADER) === -1) hdrs.unshift(KEY_COL_HEADER);
    sh.getRange(1, 1, 1, hdrs.length).setValues([hdrs]);
  }

  // Current headers
  var headers = sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0].map(function(h){
    return h == null ? '' : String(h).trim();
  });
  var keyIdx = headers.indexOf(KEY_COL_HEADER);
  if (keyIdx === -1) return respond_(e, { ok:false, error:'key-header-missing: ' + KEY_COL_HEADER });

  // Ensure columns for all payload keys
  var changed = false;
  for (var h in payload) if (payload.hasOwnProperty(h) && headers.indexOf(h) === -1) { headers.push(h); changed = true; }
  if (changed) sh.getRange(1,1,1,headers.length).setValues([headers]);

  // Find row by key
  var rowIndex = -1;
  if (sh.getLastRow() >= 2) {
    var rng = sh.getRange(2, keyIdx+1, sh.getLastRow()-1, 1).getValues();
    for (var r = 0; r < rng.length; r++) if (String(rng[r][0]) === String(key)) { rowIndex = r + 2; break; }
  }
  if (rowIndex === -1) {
    if (!ALLOW_APPEND) return respond_(e, { ok:false, error:'not-found and append-disabled: ' + key });
    rowIndex = sh.getLastRow() + 1;
  }

  // Build output row
  var out = new Array(headers.length);
  if (rowIndex <= sh.getLastRow()) {
    var cur = sh.getRange(rowIndex, 1, 1, headers.length).getValues()[0];
    for (var i=0;i<headers.length;i++) out[i] = cur[i];
  } else {
    for (var j=0;j<headers.length;j++) out[j] = '';
  }
  for (var c=0;c<headers.length;c++) {
    var H = headers[c];
    if (payload.hasOwnProperty(H)) out[c] = payload[H];
  }
  out[keyIdx] = key;

  sh.getRange(rowIndex, 1, 1, headers.length).setValues([out]);
  return respond_(e, { ok:true, row: rowIndex, key: key });
}

/** ===== IMAGE PROXY (robust) ===== */
function handleImageProxy_(e) {
  var id = e.parameter.id || '';
  var u  = e.parameter.u  || '';
  var w  = e.parameter.w  ? parseInt(e.parameter.w, 10) : 0;

  var result = _resolveImage_(id, u, w);
  if (!result.ok) return respond_(e, result);

  var blob = result.blob;
  // Ensure a browser-friendly image MIME type
  var ct = blob.getContentType();
  if (!ct || !/^image\//i.test(ct)) blob.setContentType('image/jpeg');
  return blob;
}

/** Try multiple ways to get an image; verify content-type is image/* */
function _resolveImage_(id, u, w) {
  // 1) DriveApp (needs deployer access)
  if (id) {
    try {
      var file = DriveApp.getFileById(id);
      var blob = file.getBlob();
      blob = _maybeResize_(blob, w);
      if (_isImageBlob_(blob)) return { ok:true, path:'DriveApp', blob:blob };
    } catch (err) {
      // fall through
    }
    // 2) Public download
    var url1 = 'https://drive.google.com/uc?export=download&id=' + encodeURIComponent(id);
    var r1 = _fetchImage_(url1);
    if (r1) { r1 = _maybeResize_(r1, w); if (_isImageBlob_(r1)) return { ok:true, path:'uc-download', blob:r1 }; }

    // 3) Public view
    var url2 = 'https://drive.google.com/uc?export=view&id=' + encodeURIComponent(id);
    var r2 = _fetchImage_(url2);
    if (r2) { r2 = _maybeResize_(r2, w); if (_isImageBlob_(r2)) return { ok:true, path:'uc-view', blob:r2 }; }

    // 4) Thumbnail (lets Google do the resize)
    var sz = w && w>0 ? 'w' + w : 'w800';
    var url3 = 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(id) + '&sz=' + sz;
    var r3 = _fetchImage_(url3);
    if (r3) { if (_isImageBlob_(r3)) return { ok:true, path:'thumbnail', blob:r3 }; }

    return { ok:false, error:'drive-image-not-accessible' };
  }

  // External URL path
  if (u) {
    var r = _fetchImage_(u);
    if (r) {
      r = _maybeResize_(r, w);
      if (_isImageBlob_(r)) return { ok:true, path:'urlfetch', blob:r };
    }
    return { ok:false, error:'upstream-not-image' };
  }

  return { ok:false, error:'missing id or u' };
}

function _fetchImage_(url) {
  try {
    var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true });
    var code = res.getResponseCode();
    if (code >= 400) return null;
    var blob = res.getBlob();
    // If Drive gave us HTML, contentType may be text/html — we’ll detect later.
    return blob;
  } catch (e) { return null; }
}

function _maybeResize_(blob, w) {
  if (!w || w <= 0) return blob;
  try {
    var img = ImagesService.openImage(blob).resize(w);
    return img.getBlob();
  } catch (e) { return blob; }
}
function _isImageBlob_(blob) {
  var ct = blob && blob.getContentType ? blob.getContentType() : '';
  return !!ct && /^image\//i.test(ct);
}

/** ===== DEBUG: tell me which path was used (no binary) ===== */
function imgDebug_(e) {
  var id = e.parameter.id || '';
  var u  = e.parameter.u  || '';
  var w  = e.parameter.w  ? parseInt(e.parameter.w, 10) : 0;

  // Probe without returning the blob
  var info = { ok:false, tried:[] };

  // DriveApp
  if (id) {
    try {
      var file = DriveApp.getFileById(id);
      var b = file.getBlob();
      if (_isImageBlob_(b)) return respond_(e, { ok:true, path:'DriveApp', contentType:b.getContentType() });
      info.tried.push('DriveApp-not-image');
    } catch (err) { info.tried.push('DriveApp-error'); }

    // uc download
    var url1 = 'https://drive.google.com/uc?export=download&id=' + encodeURIComponent(id);
    var r1 = _fetchImage_(url1);
    if (r1 && _isImageBlob_(r1)) return respond_(e, { ok:true, path:'uc-download', contentType:r1.getContentType() });
    info.tried.push('uc-download-' + (r1 ? r1.getContentType() : 'null'));

    // uc view
    var url2 = 'https://drive.google.com/uc?export=view&id=' + encodeURIComponent(id);
    var r2 = _fetchImage_(url2);
    if (r2 && _isImageBlob_(r2)) return respond_(e, { ok:true, path:'uc-view', contentType:r2.getContentType() });
    info.tried.push('uc-view-' + (r2 ? r2.getContentType() : 'null'));

    // thumbnail
    var sz = w && w>0 ? 'w' + w : 'w800';
    var url3 = 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(id) + '&sz=' + sz;
    var r3 = _fetchImage_(url3);
    if (r3 && _isImageBlob_(r3)) return respond_(e, { ok:true, path:'thumbnail', contentType:r3.getContentType() });
    info.tried.push('thumbnail-' + (r3 ? r3.getContentType() : 'null'));

    return respond_(e, { ok:false, error:'drive-image-not-accessible', details:info });
  }

  if (u) {
    var r = _fetchImage_(u);
    if (r && _isImageBlob_(r)) return respond_(e, { ok:true, path:'urlfetch', contentType:r.getContentType() });
    return respond_(e, { ok:false, error:'upstream-not-image', contentType: r ? r.getContentType() : 'null' });
  }

  return respond_(e, { ok:false, error:'missing id or u' });
}

/** ===== HTML Image Preview Page ===== */
function imgPage_(e) {
  var id = e.parameter.id || '';
  var w  = e.parameter.w ? parseInt(e.parameter.w,10) : 800;
  var src = '?id=' + encodeURIComponent(id) + '&w=' + w;
  var html = '<!doctype html><meta name="viewport" content="width=device-width,initial-scale=1">'
    + '<style>body{font-family:sans-serif;padding:16px}img{max-width:100%;height:auto;display:block;}</style>'
    + '<h3>Image test</h3>'
    + '<p>Below is the image served by <code>?fn=img</code>:</p>'
    + '<img src="' + src + '">';
  return HtmlService.createHtmlOutput(html);
}

/** ===== PHOTO UPLOAD ===== */
function handlePhotoUpload_(e) {
  var data = parsePayload_(e);
  if (!data || !data.photo) return respond_(e, { ok:false, error:'missing-photo' });

  var folderId = data.folderId || PHOTO_FOLDER_ID;
  var folder = null;
  try { folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder(); }
  catch (err) { return respond_(e, { ok:false, error:'invalid-folder' }); }

  var parsed = parseDataUrl_(data.photo);
  if (!parsed) return respond_(e, { ok:false, error:'invalid-data-url' });

  var name = data.name || ('upload_' + new Date().toISOString().replace(/[:.]/g,'-') + parsed.ext);
  var blob = Utilities.newBlob(parsed.bytes, parsed.mime, name);

  var file = folder.createFile(blob);
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (err) {}

  return respond_(e, { ok:true, fileId: file.getId(), name: file.getName() });
}

/** ===== Helpers ===== */
function parsePayload_(e) {
  if (e && e.parameter && e.parameter.payload) { try { return JSON.parse(e.parameter.payload); } catch (err1) {} }
  if (e && e.postData && e.postData.contents) {
    try {
      var type = String(e.postData.type || '');
      if (type.indexOf('application/json') !== -1) return JSON.parse(e.postData.contents);
      var m = /(?:^|&)payload=([^&]+)/.exec(e.postData.contents);
      if (m) return JSON.parse(decodeURIComponent(m[1]));
    } catch (err2) {}
  }
  return null;
}
function getKey_(payload) {
  return payload[KEY_COL_HEADER] ||
         payload['StableID'] ||
         payload['stable_id'] ||
         payload['stableId'] ||
         payload['key'] || '';
}
function parseDataUrl_(s) {
  if (typeof s !== 'string') return null;
  var m = /^data:([^;,]+);base64,([a-z0-9+\/=\s]+)$/i.exec(s);
  if (!m) return null;
  var mime = m[1];
  var b64  = m[2].replace(/\s+/g,'');
  var bytes = Utilities.base64Decode(b64);
  var ext = (mime === 'image/png') ? '.png'
          : (mime === 'image/gif') ? '.gif'
          : '.jpg';
  return { mime: mime, ext: ext, bytes: bytes };
}
function getSheet_() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sh = ss.getSheetByName(SHEET_TAB);
  return sh || null;
}

  DriveApp.getRootFolder();
  UrlFetchApp.fetch('https://example.com');
}
