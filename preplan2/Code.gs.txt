function doGet(e) {
  var p  = (e && e.parameter) || {};
  var fn = String(p.fn || '').toLowerCase();

  // UI route (optional)
  if (String(p.ui || '') === 'edit') return renderEditorUI_(p.key || '');

  // API routes
  if (fn === 'img')  return serveImage_(p);       // by id or u
  if (fn === 'meta') return json_({ ok: true, ts: new Date().toISOString() });
  if (fn === 'row')  return getRow_(p);
  if (fn === 'list') return listRows_(p);

  // default
  return json_({ ok: true, hello: 'HFD' });
}

function doPost(e) {
  var p = (e && e.parameter) || {};
  var fn = String(p.fn || '').toLowerCase();

  if (fn === 'save')        return saveRecord_(e);
  if (fn === 'uploadphoto') return uploadPhoto_(e);

  return json_({ error: 'unknown fn' }, 404);
}

/* ---------------- Binary / JSON outputs with headers ---------------- */

function binOut_(blob, cacheSeconds) {
  var out = ContentService.createBinaryOutput().setContent(blob.getBytes());
  out.setMimeType(blob.getContentType() || 'image/jpeg');
  out.setHeader('Access-Control-Allow-Origin', '*');
  out.setHeader('Cache-Control', 'public, max-age=' + (cacheSeconds || 3600));
  return out;
}

function json_(obj, code) {
  var out = ContentService.createTextOutput(JSON.stringify(obj || {}))
    .setMimeType(ContentService.MimeType.JSON);
  if (code) out.setResponseCode(code);
  out.setHeader('Access-Control-Allow-Origin', '*');
  out.setHeader('Access-Control-Allow-Methods', 'GET,POST,OPTIONS');
  out.setHeader('Cache-Control', 'no-store, max-age=0');
  return out;
}

/* ---------------- Image proxy (Drive id OR remote URL) ---------------- */

function serveImage_(p) {
  var id = p.id || p.fileId || '';
  if (id) return serveDriveImage_(String(id), p);

  // Remote URL: locked down with allow-list
  if (String(p.fn || '').toLowerCase() === 'img' && p.u) return serveRemoteUrl_(p);

  return json_({ error: 'missing id or u' }, 400);
}

function serveDriveImage_(id, p) {
  var w = clampWidth_(p.w);
  try {
    var file = DriveApp.getFileById(id);
    var blob = file.getBlob();
    var ct   = blob.getContentType() || 'image/jpeg';

    // Non-image Google files (Docs/Slides): try Drive thumbnail first
    if (!/^image\//i.test(ct)) {
      var tUrl = 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(id) + '&sz=w' + w;
      var tRes = UrlFetchApp.fetch(tUrl, { muteHttpExceptions: true, followRedirects: true });
      if (tRes.getResponseCode() >= 200 && tRes.getResponseCode() < 300) {
        var tb = tRes.getBlob();
        tb.setContentType('image/jpeg');
        return binOut_(tb, 3600);
      }
      // Last resort: lh3 (may 403/429 depending on file/acl)
      var gUrl = 'https://lh3.googleusercontent.com/d/' + encodeURIComponent(id) + '=w' + w;
      var gRes = UrlFetchApp.fetch(gUrl, { muteHttpExceptions: true, followRedirects: true });
      if (gRes.getResponseCode() >= 200 && gRes.getResponseCode() < 300) {
        var gb = gRes.getBlob();
        gb.setContentType('image/jpeg');
        return binOut_(gb, 1800);
      }
      return json_({ error: 'thumbnail not available' }, 404);
    }

    // Regular image: return original
    blob.setContentType(ct);
    return binOut_(blob, 86400); // cache 24h

  } catch (err) {
    return json_({ error: 'drive-fetch-failed', detail: String(err) }, 500);
  }
}

function serveRemoteUrl_(p) {
  var u = String(p.u || '');
  if (!/^https?:\/\//i.test(u)) {
    return json_({ error: 'invalid url' }, 400);
  }
  if (!isAllowedHost_(u)) {
    return json_({ error: 'host not allowed' }, 403);
  }
  try {
    var r  = UrlFetchApp.fetch(u, { muteHttpExceptions: true, followRedirects: true });
    var sc = r.getResponseCode();
    if (sc < 200 || sc >= 300) return json_({ error: 'fetch-failed', status: sc }, sc);

    var b = r.getBlob();
    b.setContentType(b.getContentType() || 'image/jpeg');
    return binOut_(b, 3600);

  } catch (err) {
    return json_({ error: 'remote-fetch-error', detail: String(err) }, 500);
  }
}

/* ---------------- Sheet helpers (wired) ---------------- */

function getSheet_() {
  try { return SpreadsheetApp.openById(CFG.SHEET_ID).getSheetByName(CFG.SHEET_TAB); }
  catch (e) { return null; }
}

function listRows_(p) {
  var sh = getSheet_();
  if (!sh) return json_({ error: 'sheet-not-found' }, 500);
  var rng = sh.getDataRange().getValues();
  if (!rng || rng.length < 2) return json_({ data: [] });

  var headers = rng[0];
  var out = [];
  for (var r = 1; r < rng.length; r++) {
    var row = {};
    for (var c = 0; c < headers.length; c++) row[headers[c]] = rng[r][c];
    out.push(row);
  }
  return json_({ data: out });
}

function getRow_(p) {
  var sh = getSheet_();
  if (!sh) return json_({ error: 'sheet-not-found' }, 500);
  var key = (p.key || '').toString().trim();
  if (!key) return json_({ error: 'missing key' }, 400);

  var rng = sh.getDataRange().getValues();
  var headers = rng[0];
  var keyIdx = headers.indexOf(CFG.KEY_HEADER);
  if (keyIdx < 0) return json_({ error: 'key-header-missing: ' + CFG.KEY_HEADER }, 500);

  for (var r = 1; r < rng.length; r++) {
    if (String(rng[r][keyIdx]) === key) {
      var rec = {};
      for (var c = 0; c < headers.length; c++) rec[headers[c]] = rng[r][c];
      return json_({ data: rec });
    }
  }
  return json_({ error: 'not-found', key: key }, 404);
}

function saveRecord_(e) {
  var sh = getSheet_();
  if (!sh) return json_({ error: 'sheet-not-found' }, 500);

  var payload = parsePayload_(e);
  if (!payload || typeof payload !== 'object') return json_({ error: 'invalid payload' }, 400);

  var rng = sh.getDataRange().getValues();
  if (!rng || rng.length === 0) {
    // Initialize headers if the sheet was empty
    var headersI = Object.keys(payload);
    sh.getRange(1, 1, 1, headersI.length).setValues([headersI]);
    rng = sh.getDataRange().getValues();
  }

  var headers = rng[0];
  var keyIdx = headers.indexOf(CFG.KEY_HEADER);
  if (keyIdx < 0) return json_({ error: 'key-header-missing: ' + CFG.KEY_HEADER }, 500);

  // Find existing row by key
  var key = (payload[CFG.KEY_HEADER] || '').toString().trim();
  var rowIndex = -1;
  for (var r = 1; r < rng.length; r++) {
    if (String(rng[r][keyIdx]) === key && key) { rowIndex = r + 1; break; } // 1-based
  }

  // Ensure all headers from payload exist
  var changed = false;
  for (var h in payload) {
    if (headers.indexOf(h) === -1) { headers.push(h); changed = true; }
  }
  if (changed) {
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  }

  // Write row
  var row = new Array(headers.length);
  for (var c = 0; c < headers.length; c++) {
    var h = headers[c];
    row[c] = payload.hasOwnProperty(h) ? payload[h] : '';
  }

  if (rowIndex === -1) {
    rowIndex = sh.getLastRow() + 1;
  }
  sh.getRange(rowIndex, 1, 1, headers.length).setValues([row]);

  return json_({ ok: true, rowIndex: rowIndex });
}

/* ---------------- Photo Upload per Column ---------------- */

function uploadPhoto_(e) {
  var sh = getSheet_();
  if (!sh) return json_({ error: 'sheet-not-found' }, 500);

  var p = parsePayload_(e) || (e && e.parameter) || {};
  var key    = (p.key || '').toString().trim();
  var column = (p.column || '').toString().trim();

  if (!key)    return json_({ error: 'missing key' }, 400);
  if (!column) return json_({ error: 'missing column' }, 400);

  var folderId = resolveFolderIdForColumn_(column);
  if (!folderId) return json_({ error: 'unknown column or folder not set', column: column }, 400);

  // Build a blob from one of: dataUrl | fileId | u
  var blob = null, sourceId = null;

  if (p.dataUrl) {
    blob = parseDataUrlToBlob_(String(p.dataUrl));
    if (!blob) return json_({ error: 'invalid dataUrl' }, 400);
  } else if (p.fileId) {
    sourceId = String(p.fileId);
    try {
      var srcFile = DriveApp.getFileById(sourceId);
      blob = srcFile.getBlob(); // we'll create a new file in the column folder
    } catch (err) {
      return json_({ error: 'bad fileId', detail: String(err) }, 400);
    }
  } else if (p.u) {
    var u = String(p.u);
    if (!/^https?:\/\//i.test(u)) return json_({ error: 'invalid url' }, 400);
    if (!isAllowedHost_(u)) return json_({ error: 'host not allowed' }, 403);
    try {
      var r = UrlFetchApp.fetch(u, { muteHttpExceptions: true, followRedirects: true });
      var sc = r.getResponseCode();
      if (sc < 200 || sc >= 300) return json_({ error: 'fetch-failed', status: sc }, sc);
      blob = r.getBlob();
      if (!/^image\//i.test(blob.getContentType() || '')) {
        // still allow, but normalize to jpeg if unknown
        blob.setContentType('image/jpeg');
      }
    } catch (err) {
      return json_({ error: 'remote-fetch-error', detail: String(err) }, 500);
    }
  } else {
    return json_({ error: 'missing dataUrl|fileId|u' }, 400);
  }

  // Create file in the target folder
  var folder = DriveApp.getFolderById(folderId);
  var filename = buildPhotoFilename_(key, column, blob);
  var file = folder.createFile(blob).setName(filename);

  if (AUTO_SHARE_PUBLIC) {
    try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (_) {}
  }

  var fileId = file.getId();
  var viewUrl   = 'https://drive.google.com/uc?export=view&id=' + encodeURIComponent(fileId);
  var thumbUrl  = 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(fileId) + '&sz=w600';
  var proxyUrl  = (CFG.WEBAPP_URL || '').replace(/\/$/, '') + '?fn=img&id=' + encodeURIComponent(fileId) + '&w=800';

  // Write to the sheet cell identified by (Stable ID, column)
  var valueToWrite = (STORE_PHOTO_AS === 'id' ? fileId : viewUrl);
  var result = writePhotoToSheet_(sh, key, column, valueToWrite);

  return json_({
    ok: true,
    fileId: fileId,
    url: viewUrl,
    thumbUrl: thumbUrl,
    proxyUrl: proxyUrl,
    wroteToColumn: column,
    rowIndex: result && result.rowIndex || null
  });
}

function resolveFolderIdForColumn_(column) {
  // Allow exact match by header; optionally match case-insensitive
  var id = PHOTO_FIELD_FOLDERS[column];
  if (id) return extractIdFromUrlOrId_(id);

  // case-insensitive match
  var k = column.toLowerCase();
  for (var h in PHOTO_FIELD_FOLDERS) {
    if (h.toLowerCase() === k) return extractIdFromUrlOrId_(PHOTO_FIELD_FOLDERS[h]);
  }
  return '';
}

function writePhotoToSheet_(sh, key, column, value) {
  var rng = sh.getDataRange().getValues();
  if (!rng || rng.length === 0) {
    sh.getRange(1, 1, 1, 2).setValues([[CFG.KEY_HEADER, column]]);
    sh.getRange(2, 1, 1, 2).setValues([[key, value]]);
    return { rowIndex: 2 };
  }

  var headers = rng[0], keyIdx = headers.indexOf(CFG.KEY_HEADER);
  if (keyIdx < 0) {
    headers.unshift(CFG.KEY_HEADER);
    sh.insertColumnBefore(1);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    keyIdx = 0;
    rng = sh.getDataRange().getValues();
  }

  var colIdx = headers.indexOf(column);
  var changed = false;
  if (colIdx < 0) {
    headers.push(column);
    sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    colIdx = headers.length - 1;
    changed = true;
  }

  // Find row by Stable ID
  var rowIndex = -1;
  for (var r = 1; r < rng.length; r++) {
    if (String(rng[r][keyIdx]) === key) { rowIndex = r + 1; break; }
  }
  if (rowIndex === -1) { // append new row
    rowIndex = sh.getLastRow() + 1;
    if (rowIndex === 1) rowIndex = 2; // ensure headers exist at row 1
  }

  // Ensure sheet has enough columns after header expansion
  var lastCol = sh.getLastColumn();
  if (colIdx + 1 > lastCol) {
    sh.insertColumnsAfter(lastCol, (colIdx + 1) - lastCol);
  }

  sh.getRange(rowIndex, colIdx + 1, 1, 1).setValue(value);
  // Also ensure Stable ID in that row
  sh.getRange(rowIndex, keyIdx + 1, 1, 1).setValue(key);

  return { rowIndex: rowIndex, headerChanged: changed };
}

/* ---------------- Utilities: photo helpers ---------------- */

function parseDataUrlToBlob_(dataUrl) {
  var m = String(dataUrl).match(/^data:([^;]+);base64,(.+)$/);
  if (!m) return null;
  var mime = m[1] || 'image/jpeg';
  var bytes = Utilities.base64Decode(m[2]);
  return Utilities.newBlob(bytes, mime, 'upload');
}

function buildPhotoFilename_(key, column, blob) {
  var ext = guessExt_(blob.getContentType() || '');
  var safeKey = String(key || '').replace(/[^a-z0-9]+/gi, '_').replace(/^_+|_+$/g, '');
  var safeCol = String(column || '').replace(/[^a-z0-9]+/gi, '_').replace(/^_+|_+$/g, '');
  var ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'UTC', "yyyyMMdd-HHmmss");
  return safeKey + '__' + safeCol + '__' + ts + ext;
}

function guessExt_(ct) {
  ct = String(ct || '').toLowerCase();
  if (ct.indexOf('png') >= 0)  return '.png';
  if (ct.indexOf('webp') >= 0) return '.webp';
  if (ct.indexOf('gif') >= 0)  return '.gif';
  return '.jpg';
}

function extractIdFromUrlOrId_(val) {
  var s = String(val || '').trim();
  var m = s.match(/[?&]id=([a-zA-Z0-9_-]{10,})/); if (m) return m[1];
  m = s.match(/\/folders\/([a-zA-Z0-9_-]{10,})/); if (m) return m[1];
  m = s.match(/^([a-zA-Z0-9_-]{20,})$/);          if (m) return m[1];
  return s;
}

/* ---------------- UI renderer (optional) ---------------- */

function renderEditorUI_(key) {
  var safeKey = (key || '').replace(/[<>&]/g,'');
  var html = HtmlService.createHtmlOutput(
    '<!doctype html><meta charset="utf-8"><title>HFD Editor</title>' +
    '<style>body{font:16px/1.4 system-ui,Segoe UI,Arial;padding:24px}</style>' +
    '<h1>HFD Editor</h1><p>Key: <b>' + safeKey + '</b></p>'
  );
  html.setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  return html;
}

/* ---------------- Misc Utilities ---------------- */

function clampWidth_(w) {
  var n = Number(w);
  if (!isFinite(n) || n <= 0) n = 800;
  return Math.max(80, Math.min(2000, Math.floor(n)));
}

function isAllowedHost_(urlStr) {
  try {
    var u = new URL(urlStr);
    var host = String(u.hostname || '').toLowerCase();
    var allow = [
      'drive.google.com',
      'lh3.googleusercontent.com',
      'googleusercontent.com'
      // Add your domains here if you proxy/fetch from them:
      // 'hopkintonma.gov', 'yourcdn.example.com'
    ];
    for (var i = 0; i < allow.length; i++) {
      if (host === allow[i] || host.endsWith('.' + allow[i])) return true;
    }
  } catch (_) {}
  return false;
}

function parsePayload_(e) {
  // Accept JSON body or URL-encoded payload
  try {
    if (e && e.postData && e.postData.type &&
        e.postData.type.indexOf('application/json') === 0 && e.postData.contents) {
      return JSON.parse(e.postData.contents);
    }
  } catch (_) {}
  var p = (e && e.parameter) || {};
  if (p.payload) {
    try { return JSON.parse(p.payload); } catch (_) {}
  }
  return p && Object.keys(p).length ? p : null;
}
